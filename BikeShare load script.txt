// Load Station Info Table (End Station)

end_station_info:
LOAD
    station_id AS end_station_id,
    name AS end_station_name_dim,
    lat AS end_lat,
    lon AS end_lon,
    region_id AS end_region_id,

// Load Main Trips Table

bikeshare_trips:
LOAD
    trip_id,
    duration_sec,
    Timestamp#(Left(start_date,19),'YYYY-MM-DD hh:mm:ss') as start_date,
    Timestamp#(Left(end_date,19),'YYYY-MM-DD hh:mm:ss') as end_date,

    WeekDay(Timestamp#(Left(start_date,19),'YYYY-MM-DD hh:mm:ss')) as start_weekday,
    Hour(Timestamp#(Left(start_date,19),'YYYY-MM-DD hh:mm:ss')) as start_hour,
	Date(
  		If(
   			Timestamp#(Left(start_date,19),'YYYY-MM-DD hh:mm:ss') <= Timestamp#(Left(end_date,19),'YYYY-MM-DD hh:mm:ss'),
    		Floor(Timestamp#(Left(start_date,19),'YYYY-MM-DD hh:mm:ss')),
    		Floor(Timestamp#(Left(end_date,19),'YYYY-MM-DD hh:mm:ss'))
  )
) AS DateKey,

    start_station_id,
    end_station_id,
    bike_number,
    zip_code,
    subscriber_type,
    c_subscription_type,
    member_birth_year,
    member_gender,
    bike_share_for_all_trip,
    start_station_latitude,
    start_station_longitude,
    end_station_latitude,
    end_station_longitude
FROM [lib://DataFiles/bikeshare_trips 2.csv]
(txt, codepage is 28591, embedded labels, delimiter is ',', msq);


// Join Start Region

LEFT JOIN (start_station_info)
LOAD
    region_id AS start_region_id,

RESIDENT bikeshare_regions;

// Join End Region

LEFT JOIN (end_station_info)
LOAD
    region_id AS end_region_id,
    region_name AS end_region_name
RESIDENT bikeshare_regions;

DROP TABLE bikeshare_regions;

///$tab Master Calendar
MasterCalendar: 
Load 
date(TempDate) AS DateKey,
date(TempDate) AS DateBridgeKey,
date(TempDate)  as DisplayDate,
Day(TempDate)  As Day,
 
Dual('Q' & ceil(month(TempDate) / 3),ceil(month(TempDate) / 3)) as Quarter,
Dual('Q' & ceil(month(TempDate) / 3)&' '&Date(TempDate,'YY'),ceil(month(TempDate) / 3)&' '&QuarterStart(TempDate)) 	as QuarterYear,
 
Month(TempDate) 							as Month, 
Date(MonthStart(TempDate),'MMM-YY') 			as MonthYear,
Date(MonthStart(TempDate),'MMM-YY') 			as MonthYear2,
 
Date(MonthStart(TempDate),'MMM-YY')				as MonthYearNum, 
Year(TempDate)  As Year,
 
Left(Year(Yearname(TempDate,0,4)),5)&'-'&Right(Year(Yearname(TempDate,0,4))+1,2)  AS [Fiscal Year],
;
 
//=== Generate a temp table of dates === 
LOAD 
date(mindate + IterNo()) AS TempDate
,maxdate // Used in InYearToDate() above, but not kept 
WHILE mindate + IterNo() <= maxdate;
// LOAD 
//  date(maxdate - IterNo()+1) AS TempDate
//  ,mindate // Used in InYearToDate() above, but not kept 
// WHILE maxdate - IterNo() + 1 >= mindate;
 
//=== Get min/max dates from Field ===/
LOAD
min(FieldValue('DateKey', recno()))-1 as mindate,
max(FieldValue('DateKey', recno())) as maxdate
AUTOGENERATE FieldValueCount('DateKey');

// Add a user key (if you have subscriber id or so, use that, else combine birth year + gender as proxy)

bikeshare_trips_with_user:
LOAD *,
     member_birth_year & '_' & member_gender AS user_key;
LOAD
    trip_id,
    duration_sec,
    Timestamp#(Left(start_date,19),'YYYY-MM-DD hh:mm:ss') as start_date,
    Timestamp#(Left(end_date,19),'YYYY-MM-DD hh:mm:ss') as end_date,
    Date(
        If(
            Timestamp#(Left(start_date,19),'YYYY-MM-DD hh:mm:ss') <= Timestamp#(Left(end_date,19),'YYYY-MM-DD hh:mm:ss'),
            Floor(Timestamp#(Left(start_date,19),'YYYY-MM-DD hh:mm:ss')),
            Floor(Timestamp#(Left(end_date,19),'YYYY-MM-DD hh:mm:ss'))
        )
    ) AS DateKey,
    start_station_id,
    end_station_id,
    bike_number,
    zip_code,
    subscriber_type,
    c_subscription_type,
    member_birth_year,
    member_gender,
    bike_share_for_all_trip,
    start_station_latitude,
    start_station_longitude,
    end_station_latitude,
    end_station_longitude
FROM [lib://DataFiles/bikeshare_trips 2.csv]
(txt, codepage is 28591, embedded labels, delimiter is ',', msq);

// Create a table with the first trip date per user

FirstTripPerUser:
LOAD
    user_key,
    Min(DateKey) AS FirstTripDate
RESIDENT bikeshare_trips_with_user
GROUP BY user_key;

// Join the FirstTripDate back to main trips

LEFT JOIN (bikeshare_trips_with_user)
LOAD
    user_key,
    FirstTripDate
RESIDENT FirstTripPerUser;

DROP TABLE FirstTripPerUser;
