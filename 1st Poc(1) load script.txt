DateBridge:

    Date([COO ApprovalDate]) AS DateBridgeKey,
    'COO ApprovalDate' AS DateType,
    [Cadre_Month_Key]

    'DM RecommendDate' AS DateType,

  where not IsNull([DM RecommendDate]) ;

    'Paid AmountDate' AS DateType,
    [Cadre_Month_Key]

  Load distinct

///$tab Master Calendar
MasterCalendar: 
Load 
date(TempDate) AS DateBridgeKey,
date(TempDate)  as DisplayDate,
Day(TempDate)  As Day,
 
Dual('Q' & ceil(month(TempDate) / 3),ceil(month(TempDate) / 3)) as Quarter,
Dual('Q' & ceil(month(TempDate) / 3)&' '&Date(TempDate,'YY'),ceil(month(TempDate) / 3)&' '&QuarterStart(TempDate)) 	as QuarterYear,
 
Month(TempDate) 							as Month, 
Date(MonthStart(TempDate),'MMM-YY') 			as MonthYear,
Date(MonthStart(TempDate),'MMM-YY') 			as MonthYear2,
 
Date(MonthStart(TempDate),'MMM-YY')				as MonthYearNum, 
Year(TempDate)  As Year,
 
Left(Year(Yearname(TempDate,0,4)),5)&'-'&Right(Year(Yearname(TempDate,0,4))+1,2)  AS [Fiscal Year],
;
 
//=== Generate a temp table of dates === 
LOAD 
date(mindate + IterNo()) AS TempDate
,maxdate // Used in InYearToDate() above, but not kept 
WHILE mindate + IterNo() <= maxdate;
